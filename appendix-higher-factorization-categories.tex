\section{Higher factorization categories and chiral derived centres at \texorpdfstring{$d\geq 2$}{d>=2}}\label{app:higher-factorization-categories}

The local theorem (Theorem~\ref{thm:main-local}) computes the brane-side
derived $E_1$-Hochschild
$HH^\bullet(A^{\mathrm{cl}}_{\partial,N},A^{\mathrm{cl}}_{\partial,N})$
at the formal-Darboux vacuum
$b_p\in\mathcal{C}^{\mathrm{op}}_\partial$.  The Mixed
Holomorphic--Topological Deligne conjecture identifies the same object
with the chiral derived centre of the closed-string sector at $d=2$.
This appendix constructs the higher factorization-categorical machinery
in which both sides live: the category of mixed-HT chiral algebras at
arbitrary $d$, the chiral derived centre as a functor on it, and the
formal-Darboux stalk as a restriction functor.  At $d=1$ the
constructions reduce to the theory of chiral algebras on curves
\cite{beilinson-drinfeld-chiral}; at $d=2$ they support the
manuscript's local theorem as the formal-Darboux stalk of the
conjecture.  The stalk carries the concrete arity-two holomorphic
collision: the diagonal local-cohomology singular product of
Theorem~\ref{thm:formal-two-dimensional-holomorphic-ope}.

\subsection{\texorpdfstring{Factorization $E_d$-algebras on smooth complex $d$-folds}{Factorization Ed-algebras on smooth complex d-folds}}\label{app:hfc:fact-Ed}

Let $X$ be a smooth complex variety of complex dimension $d$.  Write
$\mathrm{Ran}(X)$ for the Ran space of finite subsets of $X$.  Let
$\mathrm{Mfd}^{\mathrm{lc}}(X)$ be the symmetric monoidal $\infty$-category
whose objects are open subsets $U\subset X$ and whose monoidal product
is disjoint union of subsets with disjoint closures.

\begin{defn}[Holomorphic factorization $E_d$-algebra]\label{app:defn:hol-fact-Ed}
A \emph{holomorphic factorization $E_d$-algebra} on $X$ is a symmetric
monoidal $\infty$-functor
\[
  \mathcal F\colon\mathrm{Mfd}^{\mathrm{lc}}(X)\longrightarrow E_d\text{-}\mathrm{Alg},
\]
i.e.\@ an assignment $U\mapsto\mathcal F(U)\in E_d\text{-}\mathrm{Alg}$,
factorisation isomorphisms
\[
  \mathcal F(U_1\sqcup\cdots\sqcup U_n)
   \;\xrightarrow{\sim}\;
   \mathcal F(U_1)\otimes\cdots\otimes\mathcal F(U_n)
\]
for finite collections of opens with pairwise disjoint closures, and
extension-by-zero structure maps
$\mathcal F(U)\to\mathcal F(V)$ for $U\subset V$ open, satisfying the
factorization axioms of Costello--Gwilliam
\cite{costello-gwilliam}*{Vol.~I, Defn.~3.1.4}.  The
\emph{holomorphic} condition is that $\mathcal F$ is locally constant in
the $C^\infty$ direction and varies holomorphically in the complex
direction: for any holomorphic embedding
$\iota\colon U\hookrightarrow V$, the structure map
$\mathcal F(\iota)$ is a quasi-isomorphism, and the family
$\mathcal F$ on $X$ is presented by a $\mathcal D$-module on
$\mathrm{Ran}(X)$.
\end{defn}

\begin{rmk}[Equivalent presentations]
A holomorphic factorization $E_d$-algebra on $X$ admits three
equivalent presentations
\cite{beilinson-drinfeld-chiral}\cite{francis-thesis}\cite{lurie-ha}*{\S 5.5}:
\begin{enumerate}
  \item \emph{Costello--Gwilliam.}  A factorization algebra in the sense
    of \cite{costello-gwilliam}*{Vol.~I, \S 3} valued in $E_d$-algebras,
    with the holomorphic locality differential.
  \item \emph{Factorisable $\mathcal D$-module on
    $\mathrm{Ran}(X)$.}  A factorisable $\mathcal D$-module
    $\mathcal A$ on $\mathrm{Ran}(X)$ valued in $E_d$-algebras.  At
    $d=1$ this is the algebro-geometric construction of chiral
    algebras on curves \cite{beilinson-drinfeld-chiral}*{\S 3}; at $d\geq 2$ it is the
    natural higher-dimensional extension whose equivalence with (1)
    and (3) is supplied by Costello--Gwilliam
    \cite{costello-gwilliam}*{Vol.~I, \S 3} and Lurie--Ayala--Francis
    \cites{ayala-francis}*{}{lurie-ha}*{\S 5.5}.
  \item \emph{Lurie--Ayala--Francis.}  An $E_d$-algebra object of the
    $\infty$-category of factorisation algebras on $X$ in the sense of
    \cites{ayala-francis}*{}{lurie-ha}*{\S 5.5.4}.
\end{enumerate}
At $d=1$ all three reduce to chiral algebras on curves in the
algebro-geometric sense of \cite{beilinson-drinfeld-chiral}*{\S 3}.
For $d\geq 2$
the manuscript uses the Costello--Gwilliam holomorphic
factorization-algebra presentation as the closed-string sector
classical observables; the global chiral $E_d$-Deligne structure is
the conjectural layer isolated in
\S\ref{ssec:open-chiral-Ed-Deligne}.
\end{rmk}

\begin{defn}[Category of holomorphic factorization $E_d$-algebras]\label{app:defn:HolFA-d-cat}
Let $\mathrm{HolFA}_d(X)$ denote the symmetric monoidal $\infty$-category
whose objects are holomorphic factorization $E_d$-algebras on $X$ in the
sense of Definition~\ref{app:defn:hol-fact-Ed}, and whose morphisms are
natural transformations of symmetric monoidal $\infty$-functors that
intertwine the factorisation isomorphisms.  Its monoidal product is
the pointwise tensor product of $E_d$-algebras.
\end{defn}

\subsection{Mixed holomorphic-topological refinement}\label{app:hfc:mixed-HT}

The manuscript's local geometry is
$X=\R^k_{\mathrm{top}}\times M_{\mathrm{hol}}$ with $M$ a smooth
complex variety of complex dimension $d$ ($k=2$, $d=2$ in the body).
The factorization algebra carries three independent locality
differentials: support locality on $\R^k$, holomorphic locality on
$M$, and topological / locally constant in $\R^k$.

\begin{defn}[Mixed-HT factorization $E_d^{\mathrm{HT}}$-algebra]\label{app:defn:mixed-HT-fact}
A \emph{mixed-HT factorization $E_d^{\mathrm{HT}}$-algebra} on
$X=\R^k_{\mathrm{top}}\times M_{\mathrm{hol}}$ is a symmetric monoidal
$\infty$-functor
\[
  \mathcal F\colon\mathrm{Mfd}^{\mathrm{lc}}(X)\longrightarrow E_d\text{-}\mathrm{Alg}
\]
satisfying the factorization axioms together with the three locality
data:
\begin{enumerate}[label=(\roman*)]
  \item \emph{Support locality on $\R^k$.}  For any compact embedding
    $K\subset U\subset\R^k$, the section $\mathcal F$ is supported on
    $K$ when restricted to $U\setminus K$ extends to zero.
  \item \emph{Topological locality on $\R^k$.}  For any open
    $U\subset\R^k$ and any inclusion $U_0\subset U$ with
    $\pi_0(U_0)\to\pi_0(U)$ a bijection, the structure map
    $\mathcal F(U_0\times B)\to\mathcal F(U\times B)$ is a
    quasi-isomorphism for every $B\subset M$.
  \item \emph{Holomorphic locality on $M$.}  Restriction to the
    holomorphic factor is a holomorphic factorization $E_d$-algebra
    in the sense of Definition~\ref{app:defn:hol-fact-Ed}.
\end{enumerate}
\end{defn}

\begin{defn}[Category $\mathrm{ChirAlg}^{\mathrm{HT}}_{k,d}(X)$]\label{app:defn:ChirAlg-HT}
Let $\mathrm{ChirAlg}^{\mathrm{HT}}_{k,d}(X)$ denote the symmetric
monoidal $\infty$-category whose objects are mixed-HT factorization
$E_d^{\mathrm{HT}}$-algebras on $X$ in the sense of
Definition~\ref{app:defn:mixed-HT-fact}, and whose morphisms intertwine
all three locality structures.  At $k=0$ this reduces to
$\mathrm{HolFA}_d(M)$; at $d=0$ it reduces to topological factorization
algebras on $\R^k$.  The manuscript's local geometry is $k=2$, $d=2$.
\end{defn}

\subsection{The chiral derived centre functor}\label{app:hfc:chiral-Z-der}

\begin{defn}[Chiral derived centre at $d=1$]\label{app:defn:Z-der-ch-d1}
For a chiral algebra $\mathcal A\in\mathrm{HolFA}_1(C)$ on a smooth
curve $C$, the \emph{chiral derived centre}
$Z^{\mathrm{der}}_{\mathrm{ch}}(\mathcal A)$ is the chiral Hochschild
cochain complex
\[
  Z^{\mathrm{der}}_{\mathrm{ch}}(\mathcal A)
   =C^\bullet_{\mathrm{ch}}(\mathcal A,\mathcal A)
   :=\mathrm{RHom}_{\mathcal A\otimes\mathcal A^{\mathrm{op}}}(\mathcal A,\mathcal A)
\]
in the chiral monoidal structure of
\cite{beilinson-drinfeld-chiral}*{\S 3.4}.  It carries a natural
$E_2$-algebra structure (chiral Deligne theorem at \(d=1\);
\cite{beilinson-drinfeld-chiral}*{\S 4}, Lurie HA
\cite{lurie-ha}*{Thm.~5.3.1.14}, Costello--Gwilliam
\cite{costello-gwilliam}*{Vol.~I, \S 6}).  This is the proved $d=1$
chiral Deligne theorem of Vol I.
\end{defn}

\begin{defn}[Chiral derived centre at $d\geq 2$]\label{app:defn:Z-der-ch-dgeq2}
For a holomorphic factorization $E_d$-algebra
$\mathcal A\in\mathrm{HolFA}_d(X)$ on a smooth complex variety $X$ of
complex dimension $d$, the \emph{chiral derived centre at $d$} is
\[
  Z^{\mathrm{der}}_{E_d^{\mathrm{ch}}}(\mathcal A)
   =C^\bullet_{\mathrm{ch}}(\mathcal A,\mathcal A)
   :=\mathrm{RHom}_{\mathcal A\otimes\mathcal A^{\mathrm{op}}}(\mathcal A,\mathcal A)
\]
in the higher chiral monoidal structure on $\mathrm{HolFA}_d(X)$.  At
$d=1$ this is Definition~\ref{app:defn:Z-der-ch-d1}.  At $d\geq 2$ the
$\mathrm{RHom}$ is taken in the $\infty$-category of factorization
$E_d$-algebra bimodules on $X$.

\emph{Status at $d\geq 2$.}  The conjectural $E_{d+1}^{\mathrm{ch}}$-structure
on $Z^{\mathrm{der}}_{E_d^{\mathrm{ch}}}(\mathcal A)$ extending the
$E_2$-structure of the $d=1$ case is the chiral $E_d$-Deligne
conjecture
(\S\ref{ssec:open-chiral-Ed-Deligne}; Lurie
\cite{lurie-ha}*{\S 5.3.1}); partial constructions are due to
Hennion (chiral Lie algebroids on smooth schemes) and Francis
(factorisable co-operads).  The Mixed Holomorphic--Topological Deligne
conjecture at $d=2$ is the corresponding statement for
$\mathrm{ChirAlg}^{\mathrm{HT}}_{2,2}(\R^2\times\C^2)$.
\end{defn}

\begin{defn}[Mixed-HT chiral derived centre]\label{app:defn:Z-der-HT}
For $\mathcal F\in\mathrm{ChirAlg}^{\mathrm{HT}}_{k,d}(X)$, the
\emph{mixed-HT chiral derived centre} is
\[
  Z^{\mathrm{der}}_{E_d^{\mathrm{HT}}}(\mathcal F)
   =C^\bullet_{\mathrm{ch},\mathrm{HT}}(\mathcal F,\mathcal F),
\]
the $\mathrm{RHom}$ of $\mathcal F$ with itself in the bimodule
$\infty$-category over $\mathcal F\otimes\mathcal F^{\mathrm{op}}$ inside
$\mathrm{ChirAlg}^{\mathrm{HT}}_{k,d}(X)$.  The mixed-HT structure on
the output is the natural lift of the $E_{d+1}$-structure expected at
$d\geq 2$ (chiral $E_d$-Deligne conjecture).
\end{defn}

\subsection{The formal-Darboux stalk functor}\label{app:hfc:darboux-stalk-functor}

For $X=\R^k_{\mathrm{top}}\times\C^d_{\mathrm{hol}}$ and a closed point
$p\in\C^d$, the formal-Darboux stalk operation is restriction to the
formal completion at $p$.

\begin{defn}[Formal-Darboux stalk functor]\label{app:defn:Darboux-stalk-functor}
The \emph{formal-Darboux stalk functor} at $p$ is the symmetric monoidal
$\infty$-functor
\[
  (-)|_{p}\colon
  \mathrm{ChirAlg}^{\mathrm{HT}}_{k,d}(\R^k\times\C^d)
   \longrightarrow
  \mathrm{ChirAlg}^{\mathrm{HT}}_{k,d}(\R^k\times\widehat{\C^d}_p),
\]
sending $\mathcal F\mapsto\mathcal F|_{\R^k\times\widehat{\C^d}_p}$, the
restriction of the factorisable structure to the formal neighbourhood
$\widehat{\C^d}_p=\operatorname{Spf}\widehat{\mathcal O}_{\C^d,p}$.  The
holomorphic Darboux theorem trivialises the germ of $\C^d$ at $p$ to
the formal symplectic polydisk in coordinates $(z_1,\ldots,z_d)$, so
the value at $p$ depends on $p$ only through its formal isomorphism
class --- equivalently, only through the formal Darboux coordinates.
\end{defn}

\begin{prop}[Equivariance under formal symplectomorphism]\label{app:prop:Darboux-equivariance}
The formal-Darboux stalk functor of
Definition~\ref{app:defn:Darboux-stalk-functor} is equivariant under the
formal symplectomorphism group $\mathrm{Symp}^{\mathrm{form}}(\C^d)$
acting on the source category by formal coordinate change at $p$ and
on the target by the same change of formal Darboux coordinates.  At
$d=2$ on the holomorphic-symplectic factor, the stalk is canonical up
to formal Darboux change.
\end{prop}

\begin{proof}
The Darboux theorem in formal coordinates trivialises any holomorphic
symplectic form to $\omega_p=\sum_i dz_{2i-1}\wedge dz_{2i}$, with the
trivialisation defined uniquely up to the formal symplectomorphism
group.  The factorisation structure pulls back along this
trivialisation by symmetric monoidal naturality.
\end{proof}

\subsection{The bulk closed-string sector as an
\texorpdfstring{$E_2^{\mathrm{HT}}$-factorization}{E2-HT-factorization}
algebra}\label{app:hfc:bulk-as-fact-cat}

\begin{constr}[Bulk classical observables on $\R^2\times\C^2$]\label{app:constr:bulk-fact-Ed}
Let $X=\R^2_{\mathrm{top}}\times\C^2_{\mathrm{hol}}$ and
$\mathfrak g=\mathfrak h\ltimes\mathfrak h^\vee_{\mathrm{cont}}[1]$ as
in \S\ref{ssec:hamiltonian-bf-algebra}.  The closed-string sector
classical observables factorization algebra
$\mathcal A^{\mathrm{cl}}_{\mathrm{bulk}}\in
\mathrm{ChirAlg}^{\mathrm{HT}}_{2,2}(X)$ is the assignment
\[
  U\times B\longmapsto
   C^\bullet_{\mathrm{CE},\mathrm{cont}}\bigl(
     \Omega^\bullet_c(U)\widehat\otimes
     \Omega^{0,\bullet}_c(B)\widehat\otimes\mathfrak g\bigr),
\]
for $U\subset\R^2$ and $B\subset\C^2$ open, with $E_2$-algebra structure
from the symmetric algebra structure of the continuous CE cochains and
factorisation isomorphisms inherited from extension-by-zero on
disjoint opens.  The three locality differentials of
Definition~\ref{app:defn:mixed-HT-fact} are the de Rham differential
on $U$, the Dolbeault differential $\bar\partial$ on $B$, and the
internal CE differential dual to the $\mathfrak g$-bracket.
\end{constr}

\begin{prop}[Formal-Darboux stalk of the bulk]\label{app:prop:bulk-stalk}
At a closed point $p\in\C^2$, the formal-Darboux stalk of
$\mathcal A^{\mathrm{cl}}_{\mathrm{bulk}}$ is the continuous CE cochain
algebra of the shifted-cotangent BF Lie algebra at the formal polydisk:
\[
  \mathcal A^{\mathrm{cl}}_{\mathrm{bulk}}|_p
   =C^\bullet_{\mathrm{CE},\mathrm{cont}}(\mathfrak g),
   \qquad
   \mathfrak g=\mathfrak h\ltimes\mathfrak h^\vee_{\mathrm{cont}}[1],
   \quad
   \mathfrak h=\C[[z_1,z_2]]/\C\cdot 1.
\]
This is the chart algebra
$A_{b_p}=\widehat{\mathcal O}_{\C^2,p}\cong\C[[z_1,z_2]]$ in formal
Darboux coordinates equipped with the inherited $E_2$-algebra and
chiral factorisation structure of
Construction~\ref{app:constr:bulk-fact-Ed}.  Its arity-two singular
operation is the native diagonal singular product
\[
  \mathsf{J}_x(z)\mathsf{J}_y(w)
  \sim
  K^{(2)}_\Delta(z,w)\mathsf{J}_{[x,y]_{\mathfrak g}}(w),
  \qquad x,y\in\mathfrak g,
\]
with
\[
  K^{(2)}_\Delta=\frac{d\zeta_1d\zeta_2}{\zeta_1\zeta_2},\qquad
  \zeta_i=z_i-w_i .
\]
\end{prop}

\begin{proof}
Apply Definition~\ref{app:defn:Darboux-stalk-functor} to
Construction~\ref{app:constr:bulk-fact-Ed}: the de Rham and Dolbeault
Poincar\'e contractions on the formal polydisk reduce the compactly
supported sheaf data to the continuous CE cochains of $\mathfrak g$ at
$p$.  Extension by zero gives the regular product on disjoint formal
polydisks.  When two formal insertions collide, the Dolbeault
resolution of the diagonal leaves the local-cohomology quotient
\[
  H^2_\Delta(\C[[z_1,z_2,w_1,w_2]])\,d\zeta_1d\zeta_2,
\]
and the CE bracket of $\mathfrak g$ gives exactly the singular product
of Theorem~\ref{thm:formal-two-dimensional-holomorphic-ope}.  Thus the
formal completion retains the diagonal collision operation even though
its disjoint-open product is the extension-by-zero product.
\end{proof}

\subsection{The brane open category as factorization
\texorpdfstring{$E_1^{\mathrm{HT}}$}{E1-HT}-module}\label{app:hfc:brane-as-fact-cat}

\begin{defn}[Boundary open category]\label{app:defn:brane-cat}
The \emph{boundary open category} of the local theory is
\[
  \mathcal{C}^{\mathrm{op}}_\partial
   =\Phi^{\mathrm{FA}}_d(\mathcal A^{\mathrm{cl}}_{\mathrm{bulk}})\big|_{\partial X},
\]
the Costello--Gwilliam bulk--boundary restriction of the bulk
factorization algebra \cite{costello-gwilliam}*{Vol.~II} to the brane-link
boundary $\partial X$ of the Dirac brane defect
$L=\R_t\times\{p\}\subset X=\R^2_{\mathrm{top}}\times\C^2_{\mathrm{hol}}$
--- the link sphere bundle over the worldline $\R_t$, equivalently the
boundary of an excised tubular neighbourhood
$X\setminus\nu(L)$.  Its objects are factorisation $E_1$-modules over
$\mathcal A^{\mathrm{cl}}_{\mathrm{bulk}}|_{\partial X}$ (boundary
states); its morphisms are factorisation-compatible $E_1$-bimodule
morphisms.
\end{defn}

\begin{constr}[Formal-Darboux vacuum and brane $E_1$-algebra]\label{app:constr:brane-vacuum-E1}
The \emph{formal-Darboux vacuum} at the closed point $p\in\C^2$ is
the object $b_p\in\mathcal{C}^{\mathrm{op}}_\partial$ whose
endomorphism algebra is the formal-completion endomorphism algebra
\[
  A_{b_p}
   =\End_{\mathcal{C}^{\mathrm{op}}_\partial}(b_p)
   =\widehat{\mathcal O}_{\C^2,p}\cong\C[[z_1,z_2]]
\]
of the brane skyscraper $\mathcal O_p$ at $p$.  At $N$ Dirac branes
over $b_p$ the brane $E_1$-algebra of classical observables on the
worldline $\R_t$ is
\[
  A^{\mathrm{cl}}_{\partial,N}
   =C^\bullet_{\mathrm{CE}}\bigl(\mathfrak{gl}_N,\,
       \mathrm{Sym}(\mathfrak{gl}_N\oplus\mathfrak{gl}_N
                    \oplus\mathfrak{gl}_N[1])\bigr),
   \qquad
   Q\psi=[\phi_1,\phi_2],
\]
with Chan--Paton substitution $\rho_\phi\colon A_{b_p}\to M_N(\C)$,
$z_i\mapsto\phi_i$, $[\phi_1,\phi_2]=0$
(Definition~\ref{def:formal-darboux-stalk-chart}).  Curve chiral
algebras such as affine Kac--Moody, \(\beta\gamma\), Virasoro, and
\(W\)-algebras enter through a separate curve restriction and anomaly
matching before comparison with this formal disk.
\end{constr}

\begin{prop}[Brane $E_1$-Hochschild as $E_2$-algebra]\label{app:prop:brane-HH-E2}
The derived $E_1$-centre
$Z^{\mathrm{der}}_{E_1}(A^{\mathrm{cl}}_{\partial,N})$ is naturally an
$E_2$-algebra, equivalent to the Hochschild cochain complex
$HH^\bullet(A^{\mathrm{cl}}_{\partial,N},A^{\mathrm{cl}}_{\partial,N})$
with Gerstenhaber bracket and cup product.  This is Deligne's
$E_1$-Hochschild theorem
(Tamarkin~\cite{tamarkin-deligne},
McClure--Smith~\cite{mcclure-smith},
Kontsevich--Soibelman~\cite{kontsevich-soibelman-deformation},
Lurie~\cite{lurie-ha}*{Thm.~5.3.1.14}).
\end{prop}

\subsection{The Mixed Holomorphic--Topological Deligne conjecture as a
functorial statement}\label{app:hfc:mixed-HT-Deligne}

\begin{conj}[Mixed Holomorphic--Topological Deligne, functorial form]\label{app:conj:mixed-HT-Deligne}
For the bulk closed-string sector
$\mathcal A^{\mathrm{cl}}_{\mathrm{bulk}}\in\mathrm{ChirAlg}^{\mathrm{HT}}_{2,2}(\R^2\times\C^2)$
of Construction~\ref{app:constr:bulk-fact-Ed} and the boundary open
category $\mathcal{C}^{\mathrm{op}}_\partial$ of
Definition~\ref{app:defn:brane-cat}, there is a canonical equivalence
\[
  \mathcal A^{\mathrm{cl}}_{\mathrm{bulk}}
   \;\simeq\;
   Z^{\mathrm{der}}_{E_2^{\mathrm{HT}}}\bigl(\mathcal{C}^{\mathrm{op}}_\partial\bigr)
\]
of mixed-HT chiral algebras at $d=2$, where the right-hand side is the
mixed-HT chiral derived centre of the open category in the sense of
Definition~\ref{app:defn:Z-der-HT}.

\emph{Status.}
\begin{itemize}
  \item At $d=1$ (replacing $\C^2$ by a smooth curve $C$): proved
    (chiral Deligne theorem at \(d=1\),
    Vol I~\cite{lorgat-vol1}; \cite{beilinson-drinfeld-chiral}).
  \item At $d\geq 2$: open.  The chiral $E_d$-Deligne conjecture in the
    sense of Lurie~\cite{lurie-ha}*{\S 5.3.1} is the underlying purely
    chiral statement, of which the mixed-HT form is the
    holomorphic-topological enhancement.  Partial cases are recorded
    in Hennion (chiral Lie algebroids on smooth schemes) and Francis
    (factorisable co-operads).
  \item \emph{This manuscript:} proves the formal-Darboux stalk
    identification
    $\mathcal A^{\mathrm{cl}}_{\mathrm{bulk}}|_p\simeq
    Z^{\mathrm{der}}_{E_1}(A^{\mathrm{cl}}_{\partial,N})\simeq
    HH^\bullet(A^{\mathrm{cl}}_{\partial,N},
                A^{\mathrm{cl}}_{\partial,N})$
    of Conjecture~\ref{app:conj:mixed-HT-Deligne} at $N$ Dirac
    branes over the formal-Darboux vacuum $b_p$.  At the
    formal-Darboux stalk the bulk reduces to its $E_2$-algebra value
    $\mathcal A^{\mathrm{cl}}_{\mathrm{bulk}}|_p =
    C^\bullet_{\mathrm{CE},\mathrm{cont}}(\mathfrak g)$ of
    Proposition~\ref{app:prop:bulk-stalk}, and Lurie--Francis
    additivity / Costello--Gwilliam bulk--boundary correspondence
    identifies it with the brane-side derived $E_1$-centre
    $Z^{\mathrm{der}}_{E_1}(A^{\mathrm{cl}}_{\partial,N})\simeq
    HH^\bullet(A^{\mathrm{cl}}_{\partial,N},A^{\mathrm{cl}}_{\partial,N})$
    as $E_2$-algebras
    (Theorem~\ref{thm:closed-to-open-derived-centre-map}), with the
    formal-coordinate arity-two singular product computed by
    Theorem~\ref{thm:formal-two-dimensional-holomorphic-ope}.
\end{itemize}
\end{conj}

\begin{thm}[Formal-Darboux stalk of Conjecture~\ref{app:conj:mixed-HT-Deligne}]\label{app:thm:stalk-of-conj}
At $N$ Dirac branes over the formal-Darboux vacuum
$b_p\in\mathcal{C}^{\mathrm{op}}_\partial$, the formal-Darboux stalk
of Conjecture~\ref{app:conj:mixed-HT-Deligne} holds:
\[
  \mathcal A^{\mathrm{cl}}_{\mathrm{bulk}}|_p
   \;\simeq\;
   Z^{\mathrm{der}}_{E_1}\bigl(A^{\mathrm{cl}}_{\partial,N}\bigr)
   \;\simeq\;
   HH^\bullet\bigl(A^{\mathrm{cl}}_{\partial,N},A^{\mathrm{cl}}_{\partial,N}\bigr)
\]
as $E_2$-algebras, with formal Darboux coordinate expression
$J(f)=\overline{\operatorname{Tr}\,f(\phi_1,\phi_2)}$ on
$\mathfrak h=A_{b_p}/\C\cdot 1$, and reduced closed-to-open dictionary
$c_f\mapsto\theta_f\in HH^1$, $u_f\mapsto J(f)\cdot(-)\in HH^0$ on
Chevalley--Eilenberg generators of
$\mathfrak g=\mathfrak h\ltimes\mathfrak h^\vee_{\mathrm{cont}}[1]$.
The arity-two formal diagonal operation at the stalk is the singular
product of
Theorem~\ref{thm:formal-two-dimensional-holomorphic-ope}.
The chain-level lift $\Phi_N$ to a strict $P_0$-factorization-centre
morphism is conditional on
$\operatorname{Ob}_{\mathrm{cent}/\mathrm{QME}/\mathrm{desc}}=0$
(Theorem~\ref{thm:app-unreduced-cotangent-lift-obstruction}).
\end{thm}

\begin{proof}
Combine Proposition~\ref{app:prop:bulk-stalk} (formal-Darboux stalk of
the bulk),
Proposition~\ref{app:prop:brane-HH-E2} (Deligne's $E_1$-Hochschild
theorem for the brane $E_1$-algebra), and
Theorem~\ref{thm:closed-to-open-derived-centre-map} (Costello--Gwilliam
bulk--boundary correspondence at the formal-Darboux stalk).  The
trace-map computation is
Theorem~\ref{thm:formal-local-universal-property} (Procesi--Razmyslov +
Loday--Quillen--Tsygan); the dictionary is
Theorem~\ref{thm:reduced-ce-pv-central-operation}; the arity-two
diagonal operation is
Theorem~\ref{thm:formal-two-dimensional-holomorphic-ope}; the conditional
chain-level lift is
Theorem~\ref{thm:app-unreduced-cotangent-lift-obstruction}.
\end{proof}

\begin{rmk}[Formal-Darboux stalk and global conjecture]
Theorem~\ref{app:thm:stalk-of-conj} computes the formal-Darboux stalk
of Conjecture~\ref{app:conj:mixed-HT-Deligne} at the
formal-Darboux vacuum $b_p$.  The conjecture itself --- the global equivalence on
$\R^2\times\C^2$ between the bulk
$\mathcal A^{\mathrm{cl}}_{\mathrm{bulk}}$ and the mixed-HT chiral
derived centre of the boundary open category at $d=2$ --- remains
open.  The $d=2$ chiral $E_d$-Deligne theorem on which it depends is
itself open (\S\ref{ssec:open-chiral-Ed-Deligne}).  Identification
with any curve chiral algebra requires a separate restriction theorem
and matched-conventions transfer
(Theorem~\ref{thm:matched-conventions-transfer}).
\end{rmk}

\subsection{The chiral \texorpdfstring{$E_d$}{Ed}-operad: holomorphic little disks with principal-parts data}\label{app:hfc:chiral-Ed-operad}

The functorial Mixed-HT Deligne conjecture
(Conjecture~\ref{app:conj:mixed-HT-Deligne}) requires an operadic
action: the chiral derived centre
$Z^{\mathrm{der}}_{E_d^{\mathrm{ch}}}(\mathcal A)$ is the endomorphism
object in a chiral $E_d$-module category.  At $d=1$ the chiral operad
on curves supplies this structure
\cite{beilinson-drinfeld-chiral}*{\S 3.4}.  At $d\geq 2$ the operad
is controlled by higher-codimension diagonal local cohomology.  This
subsection fixes the formal operadic model used by the manuscript; at
$d=2$ its arity-two coefficient is the module computed in
Theorem~\ref{thm:formal-two-dimensional-holomorphic-ope}.

\begin{defn}[Configuration $\mathcal D$-module]\label{app:defn:conf-D-module}
Let $X$ be a smooth complex variety of complex dimension $d$ and $I$
a non-empty finite set.  Write $X^I$ for the $|I|$-fold product and
$j_I\colon\mathrm{Conf}_I(X)\hookrightarrow X^I$ for the open embedding
of the configuration space of $|I|$ distinct points.  The
\emph{configuration $\mathcal D$-module}
\[
  \omega^{\mathrm{ch}}_{X^I}=j_{I,*}j_I^!\omega_{X^I}
\]
is the $!$-extension to $X^I$ of the dualising sheaf of
$\mathrm{Conf}_I(X)$.
\end{defn}

\begin{defn}[Formal chiral $E_d$ operadic model]\label{app:defn:chiral-Ed-operad}
The \emph{formal chiral $E_d$ operadic model} on $\C^d$ is the colored $\infty$-operad
$\mathsf{hE}_d^{\mathrm{ch}}$ with single colour and operation space at
arity $|I|$
\[
  \mathsf{hE}_d^{\mathrm{ch}}(I)
   =\omega^{\mathrm{ch}}_{(\C^d)^I},
\]
with composition the chiral product on $!$-pull-back along the
diagonal embedding
$\Delta_A\colon(\C^d)^A\hookrightarrow(\C^d)^I$ for a partition
$I=\bigsqcup_{a\in A}I_a$, following
\cite{beilinson-drinfeld-chiral}*{\S 3.4} as the formal
higher-codimension Ran-space analogue.  At \(d\geq2\) this definition
is a model for the conjectural chiral \(E_d\)-operadic action.  The
full global Deligne theorem is the separate conjectural equivalence
stated below.
\end{defn}

\begin{rmk}[Comparison with curves at \texorpdfstring{$d=1$}{d=1}]\label{app:rmk:hEd-vs-BD-d1}
For $d=1$ on a curve $C$, the configuration $\mathcal D$-module
$\omega^{\mathrm{ch}}_{C^I}$ is the chiral operad on curves of
\cite{beilinson-drinfeld-chiral}*{\S 3.4}.  At $d\geq 2$ the diagonals
$\Delta_{ij}\subset(\C^d)^I$ have complex codimension $d$, so the
principal parts on $\omega^{\mathrm{ch}}_{(\C^d)^I}$ involve
distributions of $d$-fold polar order; this is the higher-codimension
chiral structure specific to \(d\geq2\).
\end{rmk}

\begin{rmk}[Arity two at \texorpdfstring{$d=2$}{d=2}]
For $X=\C^2$ and $I=\{1,2\}$, the singular quotient of
\(\omega^{\mathrm{ch}}_{X^I}\) along the pairwise diagonal is
\[
  H^2_\Delta(\C[[z_1,z_2,w_1,w_2]])\,d\zeta_1d\zeta_2
  =
  \bigoplus_{a,b\ge0}\C[[w_1,w_2]]\rho^\Delta_{a,b}.
\]
The generator \(\rho^\Delta_{0,0}=K^{(2)}_\Delta\) is the
two-dimensional Cauchy kernel.  For the Hamiltonian BF algebra, the
associated binary operation sends \(x\otimes y\) to
\([x,y]_{\mathfrak g}\), giving the formal singular product of
Theorem~\ref{thm:formal-two-dimensional-holomorphic-ope}.
\end{rmk}

intertwining the chiral product of

on curves.  At \(d=2\) and arity two, the
formal-Darboux stalk coefficient is computed unconditionally by

\end{constr}

\begin{rmk}[Status of the operation maps]
The Costello--Gwilliam and Lurie--Ayala--Francis factorization

\begin{constr}[Conditional mixed-HT enhancement

Adjoin to $\mathsf{hE}_d^{\mathrm{ch}}$ the topological $E_k$-operad

\subsection{Three hypotheses for chiral
\texorpdfstring{$E_d$}{Ed} Deligne at \texorpdfstring{$d\geq 2$}{d>=2}}\label{app:hfc:three-hypotheses}

The chiral $E_d$ Deligne identification
$C^\bullet_{\mathrm{ch}}(\mathcal A,\mathcal A)\simeq
Z^{\mathrm{der}}_{\mathsf{hE}_d^{\mathrm{ch}}}(\mathcal A)$ of
Conjecture~\ref{conj:chiral-Ed-deligne} reduces to three precise
hypotheses on $\mathsf{hE}_d^{\mathrm{ch}}$ of
Definition~\ref{app:defn:chiral-Ed-operad}.

\begin{defn}[Hypothesis F: formality of $\mathsf{hE}_d^{\mathrm{ch}}$]\label{app:hyp:formality}
The formal chiral \(E_d\) operadic model
$\mathsf{hE}_d^{\mathrm{ch}}$ is formal as an
$\infty$-operad: there is a quasi-isomorphism
\[
  \mathsf{hE}_d^{\mathrm{ch}}\;\simeq\;H_*(\mathsf{hE}_d^{\mathrm{ch}})
\]
to its homology operad in chain complexes of
$\mathcal D_{\mathrm{Ran}(\C^d)}$-modules.  At $d=1$ this is
Tamarkin's formality theorem
\cite{tamarkin-deligne}\cite{kontsevich-soibelman-deformation}; at
$d\geq 2$ it is open.
\end{defn}

\begin{defn}[Hypothesis C: centrality]\label{app:hyp:centrality}
For $\mathcal A\in\mathrm{HolFA}_d(\C^d)$, the chiral derived centre
$Z^{\mathrm{der}}_{\mathsf{hE}_d^{\mathrm{ch}}}(\mathcal A)$ is the
endomorphism object in the bimodule $\infty$-category
$\mathcal A\otimes\mathcal A^{\mathrm{op}}\text{-Mod}^{\mathrm{ch}}$:
\[
  Z^{\mathrm{der}}_{\mathsf{hE}_d^{\mathrm{ch}}}(\mathcal A)
   =\RHom_{\mathcal A\otimes\mathcal A^{\mathrm{op}}}(\mathcal A,\mathcal A).
\]
At \(d=1\) this is the chiral-centre construction for chiral algebras
on curves.
For \(d\geq2\) its topological analogue is Lurie's
\(E_d\)-centre construction \cite{lurie-ha}*{\S 5.3.1}; the
holomorphic chiral enhancement is part of the conjectural package.
\end{defn}

\begin{defn}[Hypothesis R: Ran-space descent]\label{app:hyp:ran-descent}
The factorization-homology functor
$\mathrm{Sp}^{\mathrm{ch}}_X\colon\mathrm{HolFA}_d(X)\to
\mathcal D(\mathrm{Ran}(X))\text{-Mod}$ of
Lurie--Ayala--Francis~\cite{lurie-ha}*{\S 5.5} is
Ran-cohomologically descent-faithful: for any holomorphic
factorization algebra $\mathcal A$,
\[
  \mathcal A\;\xrightarrow{\sim}\;
   \mathrm{holim}_{(I\hookrightarrow\mathrm{Ran}(X))^{\mathrm{op}}}
    \mathcal A|_I,
\]
the limit over the simplicial diagram of finite-set embeddings into
the Ran space.  At $d=1$ this is part of the factorization theorem for
chiral algebras on curves \cite{beilinson-drinfeld-chiral}*{\S 3}; at
$d\geq 2$ the corresponding
descent statement is a required input for the global chiral
\(E_d\)-Deligne theorem, with evidence from factorization homology and
Francis--Gaitsgory chiral Koszul duality
\cite{francis-gaitsgory-chiral-koszul}*{Thm.~6.5.1}.
\end{defn}

\begin{thm}[Conditional chiral $E_d$ Deligne]\label{app:thm:conditional-Ed-deligne}
Under Hypotheses~\ref{app:hyp:formality}, \ref{app:hyp:centrality},
\ref{app:hyp:ran-descent}, the chiral derived centre satisfies
\[
  Z^{\mathrm{der}}_{\mathsf{hE}_d^{\mathrm{ch}}}(\mathcal A)
   \;\simeq\;
   C^\bullet_{\mathrm{ch}}(\mathcal A,\mathcal A)
\]
as $E_{d+1}^{\mathrm{ch}}$-algebras.  Hypotheses~\ref{app:hyp:centrality}
and~\ref{app:hyp:ran-descent} are the chiral enhancements of known
topological and curve-level theorems; at \(d\geq2\) they remain part of
the conditional chiral \(E_d\)-Deligne package together with
Hypothesis~\ref{app:hyp:formality}.
\end{thm}

\begin{proof}[Proof sketch]
By Hypothesis~\ref{app:hyp:centrality} the chiral derived centre is
$\RHom_{\mathcal A\otimes\mathcal A^{\mathrm{op}}}(\mathcal A,\mathcal A)$.
By Hypothesis~\ref{app:hyp:ran-descent} this is computed by the
simplicial bar resolution of $\mathcal A$ as
$\mathcal A\otimes\mathcal A^{\mathrm{op}}$-bimodule on the Ran space,
yielding $C^\bullet_{\mathrm{ch}}(\mathcal A,\mathcal A)$.  By
Hypothesis~\ref{app:hyp:formality} the
$\mathsf{hE}_d^{\mathrm{ch}}$-action refines along the formality
quasi-isomorphism to an $E_{d+1}^{\mathrm{ch}}$-action, by the same
chain of natural transformations as Tamarkin
\cite{tamarkin-deligne} at $d=1$, after replacing the one-dimensional
chiral operad by the conditional Lurie--Ayala--Francis $\mathcal D$-module presentation
\cite{lurie-ha}*{\S 5.5.4} of the chiral operad.
\end{proof}

\begin{rmk}[Obstruction class for Hypothesis~\ref{app:hyp:formality}]\label{app:rmk:formality-obstruction}
Hypothesis~\ref{app:hyp:formality} is equivalent to the vanishing of the
Maurer--Cartan element
$\mathfrak m_d\in\mathrm{Def}\bigl(\mathsf{hE}_d^{\mathrm{ch}}\to
H_*(\mathsf{hE}_d^{\mathrm{ch}})\bigr)$ classifying the chiral
$E_d$-operad relative to its homology operad is non-trivial.  At
$d=1$ Tamarkin~\cite{tamarkin-deligne} shows $\mathfrak m_1=0$ via
the Drinfeld associator.  At $d\geq 2$ the existence of an analogous
Drinfeld associator on $\mathrm{Conf}_I(\C^d)$ for $|I|\geq 2$ is the
residual frontier.  The class
\[
  \mathrm{Ob}_F=[\mathfrak m_d]
   \in H^*\bigl(\mathrm{Def}(\mathsf{hE}_d^{\mathrm{ch}})\bigr)
\]
is the formality obstruction at $d\geq 2$.
\end{rmk}

\begin{rmk}[Chiral \(E_d\) package obstruction]\label{app:rmk:chiral-Ed-package-obstruction}
The global chiral \(E_d\)-Deligne input at \(d\geq2\) consists of the
operadic action of Definition~\ref{app:defn:chiral-Ed-operad} together
with Hypotheses~\ref{app:hyp:formality}, \ref{app:hyp:centrality}, and
\ref{app:hyp:ran-descent}.  The manuscript packages the obstruction to this
input as
\[
  \mathrm{Ob}_{\mathrm{ch}E_d}
   =
   \mathrm{Ob}_{F}\oplus\mathrm{Ob}_{C}\oplus\mathrm{Ob}_{R},
\]
where \(\mathrm{Ob}_{C}\) and \(\mathrm{Ob}_{R}\) measure the obstruction to
the chiral centrality and Ran-descent hypotheses.  The formal-Darboux
stalk theorem computes the arity-two \(d=2\) local operation on which
any global theorem restricts; vanishing of
\(\mathrm{Ob}_{\mathrm{ch}E_d}\) is the additional global condition.
\end{rmk}

\subsection{Descent over the Costello--Gwilliam brane-link boundary}\label{app:hfc:CG-descent}

The boundary open category $\mathcal{C}^{\mathrm{op}}_\partial$
(Definition~\ref{app:defn:brane-cat}) is defined globally on the
Costello--Gwilliam brane-link boundary $\partial X$.  This subsection
writes the descent spectral sequence whose \(E_2\)-collapse removes
one descent obstruction.  The global mixed-HT Deligne equivalence
further requires the other obstruction classes to vanish.

\begin{defn}[Costello--Gwilliam brane-link bundle]\label{app:defn:brane-link}
For $X=\R^2_{\mathrm{top}}\times Y_{\mathrm{hol}}$ a mixed-HT
spacetime and $L\subset X$ a Dirac brane defect, the
\emph{Costello--Gwilliam brane-link boundary} $\partial X$ is the
boundary of the excised tubular neighbourhood $\nu(L)$ of $L$ in $X$
\cite{costello-gwilliam}*{Vol.~II},
\[
  \partial X
   \;=\;\partial(X\setminus\nu(L))
   \;\cong\;
   L\times S^{c-1}_\epsilon,
\]
the unit sphere bundle of codimension $c=\mathrm{codim}_X L$ over $L$.
At $L=\R_t\times\{p\}\subset\R^2_{\mathrm{top}}\times\C^2_{\mathrm{hol}}$
the codimension is $c=5$ and
$\partial X\cong\R_t\times S^4_\epsilon$.
\end{defn}

\begin{defn}[Sheaf of factorization $E_d^{\mathrm{HT}}$-categories on $\partial X$]\label{app:defn:sheaf-fact-cat-CG}
The boundary open category extends to a sheaf
$\underline{\mathcal{C}^{\mathrm{op}}_\partial}$ of mixed-HT
factorization $E_d^{\mathrm{HT}}$-categories on $\partial X$: for
$U\subset\partial X$ open,
\[
  \underline{\mathcal{C}^{\mathrm{op}}_\partial}(U)
   =\mathcal A^{\mathrm{cl}}_{\mathrm{bulk}}|_U
    \text{-Mod}^{\mathrm{ch},\mathrm{HT}}_\partial,
\]
the boundary factorization $E_d^{\mathrm{HT}}$-modules over the
restricted bulk; restriction to a smaller open is restriction of
factorization modules.  At a closed point $p\in\partial X$ the stalk is
the formal-Darboux stalk of
Definition~\ref{app:defn:Darboux-stalk-functor}.
\end{defn}

\begin{thm}[Étale descent for the boundary open
category]\label{app:thm:cg-etale-descent}
The sheaf $\underline{\mathcal{C}^{\mathrm{op}}_\partial}$ of
Definition~\ref{app:defn:sheaf-fact-cat-CG} satisfies étale descent on
the brane-link bundle $\partial X$: for any étale cover
$\{U_a\to\partial X\}$,
\[
  \underline{\mathcal{C}^{\mathrm{op}}_\partial}(\partial X)
   \;\xrightarrow{\sim}\;
   \mathrm{holim}\bigl(\textstyle\prod_a
    \underline{\mathcal{C}^{\mathrm{op}}_\partial}(U_a)
    \rightrightarrows
    \prod_{a,b}\underline{\mathcal{C}^{\mathrm{op}}_\partial}(U_a\cap U_b)
    \rightrightarrows\cdots\bigr).
\]
\end{thm}

\begin{proof}
The Costello--Gwilliam bulk--boundary correspondence
\cite{costello-gwilliam}*{Vol.~II, \S 4} presents
$\mathcal A^{\mathrm{cl}}_{\mathrm{bulk}}|_{\partial X}$ as a
factorization algebra on $\partial X$; factorization
$\infty$-categories of modules satisfy étale descent by
\cite{costello-gwilliam}*{Vol.~II, Thm.~4.6.1}.  The factorization
$E_d^{\mathrm{HT}}$-module condition is preserved under étale pullback
because the structure maps of Definition~\ref{app:defn:mixed-HT-fact}
are local on $\partial X$.  Closure under $\mathrm{holim}$ is the
$\infty$-categorical limit closure of factorization
$\infty$-categories \cite{lurie-ha}*{\S 5.5.4.10}.
\end{proof}

\begin{thm}[Descent spectral sequence]\label{app:thm:descent-ss}
Choose an étale cover $\{U_a\to\partial X\}_{a\in A}$ trivialising the
local formal-Darboux structure.  There is a spectral sequence
\[
  E_2^{p,q}
   =H^p\bigl(\partial X,
   \mathcal H^q(\underline{\mathcal{C}^{\mathrm{op}}_\partial})\bigr)
   \;\Rightarrow\;
   H^{p+q}\bigl(\mathcal{C}^{\mathrm{op}}_\partial\bigr),
\]
where $\mathcal H^q$ is the $q$-th derived stalk.  At a formal-Darboux
point $p\in\partial X$ the stalk
$\mathcal H^*(\underline{\mathcal{C}^{\mathrm{op}}_\partial})_p$ is the
formal-Darboux stalk of Theorem~\ref{app:thm:stalk-of-conj}.
\end{thm}

\begin{proof}
Standard sheaf-cohomology descent spectral sequence for the
$\infty$-categorical sheaf of
Definition~\ref{app:defn:sheaf-fact-cat-CG}, via \v{C}ech cohomology
with respect to the étale cover and convergence by
Theorem~\ref{app:thm:cg-etale-descent}.
\end{proof}

\begin{defn}[Formal-Darboux \(E_2\)-collapse]\label{app:defn:heisenberg-collapse}
The descent spectral sequence of Theorem~\ref{app:thm:descent-ss}
\emph{collapses at the formal-Darboux strata} if the differential
\[
  d_2\colon
   H^p\bigl(\partial X,\mathcal H^q\bigr)
   \longrightarrow
   H^{p+2}\bigl(\partial X,\mathcal H^{q-1}\bigr)
\]
vanishes at every formal-Darboux stratum.  Equivalently, the
formal-Darboux stalks glue across the brane-link bundle along the
local symplectic-form trivialisation of
Proposition~\ref{app:prop:Darboux-equivariance} with trivial higher
correction.
\end{defn}

\begin{thm}[Descent obstruction at the formal-Darboux strata]\label{app:thm:global-stalk-from-collapse}
The formal-Darboux \(E_2\)-collapse condition of
Definition~\ref{app:defn:heisenberg-collapse} is precisely the
vanishing of the first descent obstruction to gluing the
formal-Darboux stalks across the Costello--Gwilliam brane-link
boundary.  The obstruction is
\[
  \mathrm{Ob}_{\mathrm{desc}}
   =[d_2]
   \in H^2\bigl(\partial X,
       \mathrm{Hom}^{-1}(\mathcal H^*,\mathcal H^*)\bigr).
\]
When this class vanishes, the local stalk theorem survives this
descent test.  The global equivalence
\[
  \mathcal A^{\mathrm{cl}}_{\mathrm{bulk}}
   \simeq
   Z^{\mathrm{der}}_{E_2^{\mathrm{HT}}}
   \bigl(\mathcal{C}^{\mathrm{op}}_\partial\bigr)
\]
still requires the remaining chiral \(E_d\), quantum master equation,
and global comparison obstructions to vanish.
\end{thm}

\begin{proof}
By Theorem~\ref{app:thm:descent-ss}, the first possible obstruction
to gluing the derived stalks through the chosen étale cover of
$\partial X$ is the
\(d_2\)-differential.  Its cohomology class is the displayed
\(\mathrm{Ob}_{\mathrm{desc}}\).  Vanishing of this class is exactly
the stated \(E_2\)-collapse condition.  Theorem~\ref{app:thm:stalk-of-conj}
identifies each formal-Darboux stalk.  Identification of the global
bulk with the global chiral derived centre requires the remaining
obstruction classes to vanish.
\end{proof}

\subsection{All-loop QME for finite-jet PVA}\label{app:hfc:all-loop-qme}

The classical PVA Jacobi identity holds at the formal-Darboux stalk
(Theorem~\ref{thm:reduced-ce-pv-central-operation}) and is the Jacobi
identity for the native two-dimensional singular product of
Theorem~\ref{thm:formal-two-dimensional-holomorphic-ope}.  Lifting the PVA
structure to all $\hbar$-orders --- the quantum master equation for
the finite-jet PVA --- requires four independent analytic /
homological inputs (KZ analytic SDR, image-choice Stokes, reflected
weights, and the stress-tensor lift $T=[Q_{\mathrm{tot}},G]$), and
remains conditional on them.  The four hypotheses below isolate the
analytic / homological control needed for all-loop convergence.

\paragraph{Convention.}
Under the binding convention of
\nameref{ssec:binding-convention-layer}, the symbol
$C^\bullet_{\mathrm{ch}}(A_b,A_b)$ in this subsection denotes the
brane-side derived $E_1$-Hochschild
$HH^\bullet(A^{\mathrm{cl}}_{\partial,N},A^{\mathrm{cl}}_{\partial,N})
\simeq Z^{\mathrm{der}}_{E_1}(A^{\mathrm{cl}}_{\partial,N})$
identified with the bulk-stalk
$\mathcal A^{\mathrm{cl}}_{\mathrm{bulk}}|_b$ by Costello--Gwilliam
bulk--boundary correspondence (Theorem~\ref{thm:main-local});
$Q_{\mathrm{ch}}$ is the chiral cochain differential inherited from
$\mathcal A^{\mathrm{cl}}_{\mathrm{bulk}}|_b$ under this
identification.  The hypotheses below are formulated on that object.

\begin{defn}[Hypothesis K: KZ analytic SDR]\label{app:hyp:KZ-SDR}
There is a strong deformation retract
\[
  \bigl(C^\bullet_{\mathrm{CE},\mathrm{cont}}(\mathfrak g),Q_{\mathrm{tot}}\bigr)
   \xrightarrow[\text{retract}]{\text{deform}}
   \bigl(C^\bullet_{\mathrm{ch}}(A_b,A_b),Q_{\mathrm{ch}}\bigr)
\]
in the category of chain complexes over the analytic ring
$\C\{\!\!\{\hbar\}\!\!\}$; the formal ring $\C[[\hbar]]$ is replaced by
this analytic coefficient ring,
with finite-rank cohomology in each PVA weight, and with retract
data analytic in $\hbar$ in a punctured neighbourhood of $\hbar=0$.
\end{defn}

\begin{rmk}[Analytic SDR]\label{app:rmk:why-analytic}
The classical PVA Jacobi holds in the $\hbar$-adic topology of the
formal disk.  All-loop lifting requires Stokes-controlled analytic
continuation across the disk
(Hypothesis~\ref{app:hyp:stokes}).  A purely $\hbar$-formal SDR gives
finite-loop existence (Costello~\cite{costello-renormalization}*{Ch.~5});
analytic SDR is needed for all-loop convergence.
\end{rmk}

\begin{defn}[Hypothesis S: Stokes regularity]\label{app:hyp:stokes}
The KZ connection on $C^\bullet_{\mathrm{ch}}(A_b,A_b)$ associated
to the formal \(E_2\) singular product with kernel \(K^{(2)}_\Delta\) has trivial
Stokes data along the
real-positive $\hbar$-axis: every analytic continuation across
$\arg\hbar\in(-\pi,\pi]$ identifies with a single asymptotic series
with matching sectorial asymptotics.
\end{defn}

\begin{defn}[Hypothesis W: reflected weights]\label{app:hyp:reflected-weights}
Under crossing $z\leftrightarrow w$ on the formal \(E_2\) singular product of
Theorem~\ref{thm:formal-two-dimensional-holomorphic-ope}, the PVA
weight grading on $C^\bullet_{\mathrm{ch}}(A_b,A_b)$ reflects: the
operator
\[
  R\colon\mathrm{wt}_z\otimes\mathrm{wt}_w
   \longrightarrow\mathrm{wt}_w\otimes\mathrm{wt}_z
\]
intertwines the singular product with itself; equivalently,
$C^\bullet_{\mathrm{ch}}(A_b,A_b)$ is reflection-positive in the
chiral $E_2$-sense.
\end{defn}

\begin{defn}[Hypothesis G: \texorpdfstring{$T=[Q_{\mathrm{tot}},G]$}{T = [Q,G]}]\label{app:hyp:T-Q-G}
The energy--momentum tensor $T\in C^\bullet_{\mathrm{ch}}(A_b,A_b)$
is BRST-exact with primitive $G$:
\[
  T=[Q_{\mathrm{tot}},G],
\]
where $Q_{\mathrm{tot}}=Q_{\mathrm{CE}}+\hbar Q_{\mathrm{BV}}$ is the
total BRST operator and $G$ is a degree-$(-1)$ density on the
worldvolume.  At the formal-Darboux stalk the primitive exists as a
classical limit; the hypothesis is its survival to all $\hbar$-orders.
\end{defn}

\begin{thm}[Conditional all-loop QME]\label{app:thm:conditional-qme}
Under Hypotheses~\ref{app:hyp:KZ-SDR}, \ref{app:hyp:stokes},
\ref{app:hyp:reflected-weights}, \ref{app:hyp:T-Q-G}, the
formal-Darboux stalk of Theorem~\ref{app:thm:stalk-of-conj} lifts to
all orders in $\hbar$: there exists a quantum BV master action
$S^{\mathrm{quant}}=S^{\mathrm{cl}}+\sum_{n\geq 1}\hbar^n S^{(n)}$
satisfying
$\{S^{\mathrm{quant}},S^{\mathrm{quant}}\}+\hbar\Delta S^{\mathrm{quant}}=0$
with
\[
  \mathcal A^{\mathrm{cl}}_{\mathrm{bulk}}|_p[[\hbar]]
   \;\simeq\;
   Z^{\mathrm{der}}_{E_1}\bigl(A^{\mathrm{cl}}_{\partial,N}[[\hbar]]\bigr)
\]
holding at all $\hbar$-orders, and the trace map
$J(f)=\overline{\operatorname{Tr}\,f(\phi_1,\phi_2)}$ extending to
$J^{\mathrm{quant}}(f)=J(f)+\sum_n\hbar^n J^{(n)}(f)$.
\end{thm}

\begin{proof}[Proof sketch]
Costello's renormalization theorem
\cite{costello-renormalization}*{Thm.~8.6.0.1} gives finite-loop
existence and uniqueness modulo cohomological obstructions.
Hypothesis~\ref{app:hyp:KZ-SDR} (analytic SDR) lifts existence to a
punctured analytic neighbourhood of $\hbar=0$.
Hypothesis~\ref{app:hyp:stokes} (Stokes regularity) extends existence
across the disk $|\hbar|<r_0$ for some $r_0>0$ with matching
sectorial asymptotics.  Hypothesis~\ref{app:hyp:reflected-weights} (reflected
weights) is the obstruction-vanishing input for the chiral
$E_2$-action on $C^\bullet_{\mathrm{ch}}$ at all loops.
Hypothesis~\ref{app:hyp:T-Q-G} ($T=[Q_{\mathrm{tot}},G]$) eliminates
the trace anomaly: BRST-exactness of $T$ trivialises the
conformal-anomaly cohomology class in
$H^2(C^\bullet_{\mathrm{ch}}(A_b,A_b))$ via chain homotopy.
\end{proof}

\begin{rmk}[QME obstruction class]\label{app:rmk:qme-obstruction}
The four hypotheses combine into
\[
  \mathrm{Ob}_{\mathrm{QME}}
   =[\mathfrak{ob}_K]\oplus[\mathfrak{ob}_S]\oplus[\mathfrak{ob}_W]\oplus[\mathfrak{ob}_G]
   \in H^*\bigl(\text{local functionals on }\R^2\times\C^2\bigr),
\]
with summands the analytic-SDR obstruction $[\mathfrak{ob}_K]$, the
Stokes-filtration class $[\mathfrak{ob}_S]$, the crossing class
$[\mathfrak{ob}_W]$, and the conformal-anomaly class
$[\mathfrak{ob}_G]$.  Each summand is checkable in finitely many PVA
weights at fixed loop order.
\end{rmk}

\subsection{Operator-level lift of the modular line at \texorpdfstring{$\Delta_5$}{Delta5}}\label{app:hfc:operator-lift-modular}

The Igusa cusp form $\Phi_{10}$ is the level-2 section of the modular
line $\Omega_{\mathrm{central}}$ on the Siegel period domain
$\mathfrak H_2$
\cite{igusa-cusp-form}\cite{gritsenko-nikulin-paramodular}.  The square
root $\Delta_5=\Phi_{10}^{1/2}$ is a scalar modular form in the
external \(K3\times E\) comparison problem
(\S\ref{ssec:open-modular-lift}).  The formal-Darboux theorem
constructs the local Capelli scalar.  The operator-level lift asks for
an operator whose protected trace is \(\Delta_5\) after the global
compact-target theory and its trace have been constructed.

\begin{conj}[Operator lift of $\Delta_5$]\label{app:conj:operator-lift-Delta5}
There exists a graded vertex-algebraic or factorization-algebraic
object \(V\), an involution or real form \(\sigma\), and a protected
trace such that
\[
  \mathrm{Tr}_V\bigl(\sigma\,q^{L_0-c/24}\,
   \prod_{i=1}^{2}z_i^{H_i}\bigr)
   \;=\;\Delta_5(\tau,\rho,\nu)
   \;=\;\Phi_{10}(\tau,\rho,\nu)^{1/2}
\]
on the level-$2$ cover of the Siegel period domain
$(\tau,\rho,\nu)\in\mathfrak H_2$, with the trace compatible with
the modular line \(\Omega_{\mathrm{central}}\).  This is an open
operator-construction problem extending the formal-Darboux stalk.
\end{conj}

Three scalar theories give evidence for the desired trace.  The
operator object requires an additional construction.

\begin{constr}[Borcherds--Kac--Moody candidate]\label{app:constr:BKM-candidate}
The Borcherds--Kac--Moody denominator formulas give automorphic
products, including the fake-Monster denominator and the
Gritsenko--Nikulin products
\cite{borcherds-fake}\cite{gritsenko-nikulin-paramodular}.  These are
scalar identities.  Passing from a scalar denominator formula to a
vertex-algebraic object whose graded trace is \(\Delta_5\) requires an
additional construction.

\emph{Obstruction.}
\(\mathrm{Ob}_{\mathrm{BKM}}\) is the obstruction to lifting the
scalar Borcherds product to an operator trace with the required
integral form, real structure, and modular covariance.
\end{constr}

\begin{constr}[Free-field realisation on
\texorpdfstring{$K3\times E$}{K3 times E}]\label{app:constr:K3E-candidate}
The elliptic genus of \(K3\) \cite{witten-elliptic-genus}, its
second-quantised product, and the Donaldson--Thomas /
Pandharipande--Thomas comparison on
\(K3\times E\) \cite{maulik-pandharipande-thomas} supply scalar modular data.  A free-field chiral
algebra on \(K3\times E\) whose trace is
\(\Delta_5\).

\emph{Obstruction.}
\(\mathrm{Ob}_{K3E}\) is the obstruction to constructing the compact
target factorization object, its protected trace, and the descent of
that trace to the level-\(2\) modular line.
\end{constr}

\begin{constr}[Dabholkar--Murthy--Zagier candidate]\label{app:constr:DMZ-candidate}
The Dabholkar--Murthy--Zagier decomposition of meromorphic Jacobi
forms separates polar and finite parts in the \(1/\Phi_{10}\)
dyonic counting problem \cite{dabholkar-murthy-zagier}.  This is a
statement about mock modular and wall-crossing structures.  An
operator construction of \(\Delta_5\) is a separate lift.

\emph{Obstruction.}
\(\mathrm{Ob}_{\mathrm{DMZ}}\) is the obstruction to lifting the
mock-modular wall-crossing package to an honest operator trace with
the required holomorphic modular transformation law.
\end{constr}

\begin{rmk}[Composition of operator-lift obstructions]\label{app:rmk:operator-lift-obstructions}
The three candidates are evidence for the same scalar modular form,
and the operator construction is the common problem.  The symbol
\(\mathrm{Ob}_V\) denotes the combined obstruction to constructing an
operator-level lift from any of these scalar inputs.  A cohomological
product formula for these scalar obstruction inputs would require a
separate definition.
\end{rmk}

\subsection{Convergent obstruction taxonomy}\label{app:hfc:obstruction-taxonomy}

The four conditional theorems compose to a conditional global Mixed
Holomorphic--Topological Deligne theorem.

\begin{thm}[Conditional global Mixed-HT Deligne obstruction package]\label{app:thm:conditional-global-deligne}
Under the joint vanishing
\[
  \mathrm{Ob}_{\mathrm{global}}
   :=\mathrm{Ob}_{\mathrm{ch}E_d}\oplus
     \mathrm{Ob}_{\mathrm{desc}}\oplus
     \mathrm{Ob}_{\mathrm{QME}}\oplus
     \mathrm{Ob}_{V}
   =0,
\]
with the four summands as in
Remark~\ref{app:rmk:chiral-Ed-package-obstruction},
Theorem~\ref{app:thm:global-stalk-from-collapse},
Remark~\ref{app:rmk:qme-obstruction}, and
Remark~\ref{app:rmk:operator-lift-obstructions}, the Mixed-HT
Deligne conjecture~\ref{app:conj:mixed-HT-Deligne} holds globally on
$\R^2_{\mathrm{top}}\times\C^2_{\mathrm{hol}}$ at $d=2$, $N$ Dirac
branes, provided the construction hypotheses named in those four
inputs have been supplied.  This implication is conditional.  It does
record the compact-target modular-form comparison with an operator
trace as a separate lift.  The appearance of \(\Phi_{10}\) or
\(\Phi_{10}^{1/2}\) belongs to the
separate \(K3\times E\) operator-lift problem.
\end{thm}

\begin{proof}
Compose Theorem~\ref{app:thm:conditional-Ed-deligne} (chiral $E_d$
Deligne via the operadic action and
Hypotheses~\ref{app:hyp:formality}--\ref{app:hyp:ran-descent}),
Theorem~\ref{app:thm:global-stalk-from-collapse}
(Costello--Gwilliam brane-link descent via
Definition~\ref{app:defn:heisenberg-collapse}),
Theorem~\ref{app:thm:conditional-qme} (all-loop QME via
Hypotheses~\ref{app:hyp:KZ-SDR}--\ref{app:hyp:T-Q-G}), and
Conjecture~\ref{app:conj:operator-lift-Delta5} only in the external
compact-target comparison where that datum is part of the input.  The
four obstructions are independent cohomology classes living in
distinct cohomology groups, so their joint vanishing is well-defined.
\end{proof}

\begin{rmk}[Reduction status]\label{app:rmk:reduction-status}
The obstruction $\mathrm{Ob}_V$ is the operator-level lift problem
for the modular line on $K3\times E$ (\S\ref{ssec:open-modular-lift});
the obstruction $\mathrm{Ob}_{\mathrm{QME}}$ is the all-loop quantum
master equation lifting the finite-jet PVA across all
$\hbar$-orders.  The chiral $E_d$ Deligne
package obstruction $\mathrm{Ob}_{\mathrm{ch}E_d}$ at $d\geq 2$ is the
residual research frontier of Conjecture~\ref{conj:chiral-Ed-deligne};
partial results (chiral Lie algebroids on smooth schemes via Hennion,
factorisable co-operads via Francis) give evidence for pieces of the
package; the global theorem remains the conjectural target.  The brane-link
descent obstruction $\mathrm{Ob}_{\mathrm{desc}}$ reduces to the
$E_2$-collapse condition
of Definition~\ref{app:defn:heisenberg-collapse}, checkable on a
formal-Darboux stratum-by-stratum basis.  The manuscript's main
theorem (Theorem~\ref{thm:main-local}) is the formal-Darboux stalk
that anchors all four obstructions, and
Theorem~\ref{thm:formal-two-dimensional-holomorphic-ope} supplies its
arity-two formal diagonal operation: each conditional theorem has the stalk
theorem, including this diagonal singular product, as its base case.
\end{rmk}

\begin{constr}[Global manuscript program]\label{app:constr:program}
The conditional global theorem
(Theorem~\ref{app:thm:conditional-global-deligne}) translates the
manuscript's frontier (\S\ref{sec:frontier}) into a single
cohomological vanishing problem
$\mathrm{Ob}_{\mathrm{global}}=0$.  The four summands are independent,
hence individually attackable:

\begin{itemize}
  \item $\mathrm{Ob}_{\mathrm{ch}E_d}$: existence, centrality,
    Ran-descent, and formality for the chiral \(E_d\)-operadic package
    at \(d\geq2\); its formality summand is the chiral analogue of the
    Drinfeld associator on \(\mathrm{Conf}_I(\C^d)\), \(|I|\geq2\).
  \item $\mathrm{Ob}_{\mathrm{desc}}$: vanishing of the $d_2$
    differential in the descent spectral sequence
    (Theorem~\ref{app:thm:descent-ss}) on the formal-Darboux
    stratum.
  \item $\mathrm{Ob}_{\mathrm{QME}}$: each of four checkable
    cohomology classes
    (Hypotheses~\ref{app:hyp:KZ-SDR}--\ref{app:hyp:T-Q-G}).
  \item $\mathrm{Ob}_V$: any of three candidate constructions
    (Constructions~\ref{app:constr:BKM-candidate},
    \ref{app:constr:K3E-candidate},
    \ref{app:constr:DMZ-candidate}).
\end{itemize}

The formal-Darboux stalk theorem of this manuscript
(Theorem~\ref{thm:main-local}) anchors each.
\end{constr}
