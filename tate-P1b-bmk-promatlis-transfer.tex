\subsection{The Bochner--Martinelli--Koppelman current transfer and the pro-Matlis retract}\label{ssec:bmk-promatlis-transfer}

The heat-kernel parametrix of Theorem~\ref{thm:hadamard-mittag-leffler}
and the holomorphic current sector of the formal polydisk meet in the
support-local transfer problem; the Bochner--Martinelli--Koppelman
kernel computes its obstruction.

\begin{rmk}[Hadamard parametrix versus the Bochner-Martinelli-Koppelman kernel]\label{rmk:hadamard-vs-bmk}
  The Hadamard parametrix used in
  Theorem~\ref{thm:hadamard-mittag-leffler} controls Costello's
  heat-kernel propagator \(P_{\epsilon,L}^{\mathrm{base}}\) on the bulk
  free elliptic complex
  \(\Omega^\bullet(\R^2)\widehat\otimes\Omega^{0,\bullet}(\C^2)\) by
  short-time asymptotics in Hadamard normal form, with Hörmander
  wave-front-set tracking
  \cite{costello-renormalization}*{Ch.~5},
  \cite{hormander-vol1}*{Ch.~7}.  The Bochner-Martinelli-Koppelman
  kernel
  \[
    K_{\mathrm{BM}}(\zeta,z)
      =
    \frac{1}{(2\pi i)^2}
    \frac{
      \overline{\eta_1}\,d\bar\eta_2-\overline{\eta_2}\,d\bar\eta_1
    }{r^4}
    \wedge d\zeta_1\wedge d\zeta_2,
    \qquad
    \eta_j=\zeta_j-z_j,\ r=|\zeta-z|,
  \]
  is the several-complex-variables homotopy kernel of
  \cite{ruppenthal-bmk}*{\S5} producing the classical Koppelman
  formula
  \(\bar\partial H+H\bar\partial=\operatorname{id}-i_Bp_B\) on bounded
  \(C^1\) domains
  (\cite{ruppenthal-bmk}*{Def.~5.1, Thm.~5.2, Eq.~(10)};
  Laurent-Thiébaut \cite{laurent-thiebaut-bmk}*{Ch.~III, Def.~1.2,
  Lem.~1.3}).  These are two distinct analytic technologies attached
  to two different sectors of the local theorem.  The heat-kernel
  parametrix governs the Costello/QME graph expansion of
  Theorem~\ref{thm:hadamard-mittag-leffler}; the BMK kernel governs the
  Dolbeault homotopy on the formal holomorphic polydisk
  \(\widehat{\C^2}_0\) and the principal-part current sector recorded in
  Theorem~\ref{thm:app-finite-window-bmk-current-transfer}.  The formulas
  remain separate: the Hadamard parametrix uses the bulk
  Riemannian/Hermitian heat data on \(\R^2\times\C^2\), the BMK kernel
  uses the holomorphic homotopy data on \(\C^2\) alone, and the local
  theorem combines their separate finite-window classes through the
  obstruction vectors stated below.
\end{rmk}

\begin{thm}[BMK finite-window obstruction for strict all-window
  Hamiltonian transfer]\label{thm:bmk-finite-window-obstruction}
  Let \(\mathfrak h=\C[[z_1,z_2]]/\C\cdot1\) be the Hamiltonian Lie
  algebra of the formal symplectic polydisk \((\widehat{\C^2}_0,\omega)\),
  let
  \[
    \rho_{a,b}=z_1^{-a-1}z_2^{-b-1}\,dz_1\wedge dz_2,
    \qquad
    \mc P=\bigoplus_{\substack{a,b\geq0\\a+b>0}}\C\rho_{a,b},
    \qquad
    \mc P_{\leq N}=\bigoplus_{0<a+b\leq N}\C\rho_{a,b},
  \]
  and write \(\pi_N\colon\mc P\twoheadrightarrow\mc P_{\leq N}\) for the
  finite-Matlis truncation.  For every \(N\geq1\), the following
  \(\C\)-linear quotient problem has obstruction:
  \[
    q_N\colon\mc P\longrightarrow\mc P_{\leq N}
  \]
  retains \(\mc P_{\leq N}\) pointwise, kills the high transverse
  currents \(\rho_{a,b}\) with \(a+b>N\), and intertwines the full
  Hamiltonian coadjoint action of \(\mathfrak h\) on principal parts.
  In particular, the support-local quotient problem on the
  derivative-delta-current module
  \(\Delta_{a,b}\in\mathcal D'_0{}^{0,2}(B)\) of
  Theorem~\ref{thm:app-finite-window-bmk-current-transfer} has the same
  obstruction to carrying the Hamiltonian action onto a finite Matlis
  target.  The obstruction is the
  Tate-cohomology class of the principal-part module recorded by
  \[
    \operatorname{ob}_{\mathfrak h,N}
      =\pi_N\bigl(\mathfrak h\cdot\ker\pi_N\bigr)
      \neq 0
    \quad\text{in}\quad
    \mc P_{\leq N}.
  \]
  The class is exhibited by the explicit boundary-mode identity
  \begin{equation}\label{eq:bmk-bockstein-identity}
    z_1^{N+2}\cdot\rho_{N+1,0}=-(N+2)\,\rho_{0,1},
  \end{equation}
  where the high transverse mode \(\rho_{N+1,0}\) lies in the killed
  window \(\ker\pi_N\) but the low brane-momentum mode \(\rho_{0,1}\)
  is retained in \(\mc P_{\leq N}\).  Equivalently, the projected
  finite-window \(L_\infty\)-extension has nonzero unary block
  \[
    \kappa_N(z_1^{N+2},\rho_{N+1,0})=-(N+2)\,\rho_{0,1},
  \]
  and the projected Jacobiator on the triple
  \((z_1^{N+2},z_2,\rho_{N,0})\) is
  \begin{equation}\label{eq:bmk-bockstein-jacobiator}
    (N+1)(N+2)\,\rho_{0,1}\neq 0.
  \end{equation}
  Therefore strict all-window finite-current Hamiltonian transfer
  \[
    \C[z_1,z_2]_{\mathrm{red}}\longrightarrow
    \mathrm{finite\ Matlis\ window}
  \]
  has a nonzero obstruction for every \(N\).  The
  full obstruction vector
  \(\operatorname{Ob}^{\mathrm{BMK\mbox{-}curr}}_N\) of
  Theorem~\ref{thm:app-finite-window-bmk-current-transfer}, with the
  third entry the displayed
  \(\operatorname{ob}_{\mathfrak h,N}\), is the precise finite-window
  obstruction datum.
\end{thm}

\begin{proof}
  The Hamiltonian coadjoint action of
  Corollary~\ref{cor:app-factorization-principal-part-current-brackets}
  gives, for \(f=z_1^pz_2^q\) and \(\rho_{i,j}\),
  \[
    f\cdot\rho_{i,j}=-(pj-qi+p-q)\,\rho_{i-p+1,j-q+1}.
  \]
  Specializing \((p,q)=(N+2,0)\) and \((i,j)=(N+1,0)\),
  \[
    z_1^{N+2}\cdot\rho_{N+1,0}=-(N+2)\,\rho_{(N+1)-(N+2)+1,0-0+1}
      =-(N+2)\,\rho_{0,1},
  \]
  which is~\eqref{eq:bmk-bockstein-identity}.  The mode
  \(\rho_{N+1,0}\) is killed by \(\pi_N\) since \(N+1+0>N\); the mode
  \(\rho_{0,1}\) is retained in \(\mc P_{\leq N}\) since \(0+1\leq N\)
  for every \(N\geq1\).  Hence
  \(\rho_{N+1,0}\in\ker\pi_N\) and
  \(\pi_N(z_1^{N+2}\cdot\rho_{N+1,0})=-(N+2)\,\rho_{0,1}\neq 0\), so
  \(\operatorname{ob}_{\mathfrak h,N}\neq 0\) in \(\mc P_{\leq N}\).

  A support-local current quotient \(q_N\) intertwining the
  Hamiltonian action would satisfy
  \(q_N(\mathfrak h\cdot\xi)=\mathfrak h\cdot q_N(\xi)\) for every
  \(\xi\in\mc P\); applied to \(\xi=\rho_{N+1,0}\), with
  \(q_N(\rho_{N+1,0})=0\) by the killing hypothesis, this would force
  \[
    0=\mathfrak h\cdot q_N(\rho_{N+1,0})
      =q_N(\mathfrak h\cdot\rho_{N+1,0})
      \ni q_N(z_1^{N+2}\cdot\rho_{N+1,0})
      =-(N+2)\,q_N(\rho_{0,1})
      =-(N+2)\,\rho_{0,1},
  \]
  contradicting \(N+2\neq 0\) and \(\rho_{0,1}\neq 0\) in
  \(\mc P_{\leq N}\).  This is the obstruction.

  The projected
  Jacobiator~\eqref{eq:bmk-bockstein-jacobiator} is the nonzero
  Jacobiator of the truncated bracket: the unprojected coadjoint
  bracket on
  \(\mathfrak h\ltimes\mc P[1]\) is a genuine semidirect Lie module
  and obeys Jacobi; its image after \(\pi_N\) has the displayed
  obstruction.
  Concretely, on the triple \((z_1^{N+2},z_2,\rho_{N,0})\) the three
  cyclic terms in the lifted Jacobi sum are computed by the displayed
  coadjoint rule together with the Hamiltonian Poisson rule
  \(\{H_{a,b},H_{r,s}\}=(as-br)H_{a+r-1,b+s-1}\) of the polydisk:
  \[
  \begin{aligned}
    \{z_1^{N+2},\{z_2,\rho_{N,0}\}\}
      &=\{z_1^{N+2},(N+1)\rho_{N+1,0}\}
       =-(N+1)(N+2)\,\rho_{0,1}, \\
    \{z_2,\{\rho_{N,0},z_1^{N+2}\}\}
      &=\{z_2,0\}=0
       \quad\text{since } \{\rho_{N,0},z_1^{N+2}\}
       =-(z_1^{N+2}\cdot\rho_{N,0})=0\\
      &\hspace{3.7cm}\text{by the negative-index rule of
       Proposition~\ref{prop:app-matlis-coadjoint-action}},\\
    \{\rho_{N,0},\{z_1^{N+2},z_2\}\}
      &=\{\rho_{N,0},(N+2)\,z_1^{N+1}\}
       =(N+1)(N+2)\,\rho_{0,1}.
  \end{aligned}
  \]
  The unprojected sum is zero, as required by Jacobi for the lifted
  bracket.  Apply \(\pi_N\) entrywise: the first cycle's intermediate
  \(\{z_2,\rho_{N,0}\}=(N+1)\rho_{N+1,0}\) lies in \(\ker\pi_N\) since
  \(N+1>N\), so the projected first cycle is
  \(\pi_N\{z_1^{N+2},\pi_N(N+1)\rho_{N+1,0}\}
   =\pi_N\{z_1^{N+2},0\}=0\); the second cycle is already zero; the
  third cycle is
  \(\pi_N\{\rho_{N,0},(N+2)z_1^{N+1}\}=(N+1)(N+2)\rho_{0,1}\) since
  \(\rho_{0,1}\in\mc P_{\leq N}\) is retained.  The projected sum is
  therefore \((N+1)(N+2)\rho_{0,1}\neq 0\), which
  is~\eqref{eq:bmk-bockstein-jacobiator}.  This is the projected
  escape-return Jacobiator recorded in
  Theorem~\ref{thm:app-finite-window-bmk-current-transfer} on the same
  triple; any quotient \(q_N\) making this return exact as a unary
  boundary would also collapse \(\rho_{0,1}\) and hence
  \(\mc P_{\leq N}\) itself.

  The third entry \(\operatorname{ob}_{\mathfrak h,N}\) of
  \(\operatorname{Ob}^{\mathrm{BMK\mbox{-}curr}}_N\) in
  Theorem~\ref{thm:app-finite-window-bmk-current-transfer} is by
  construction the displayed class.  The remaining entries -- diagonal
  current sign, finite-versus-analytic moment kernel, collar
  factorization compatibility -- are independent obstruction
  coordinates of the same finite-window obstruction statement and are
  recorded there.
\end{proof}

\begin{cor}[Pro-Matlis retract as the compatible replacement target]\label{cor:promatlis-retract-compatible-replacement}
  Theorem~\ref{thm:bmk-finite-window-obstruction} forecloses the fixed
  finite-Matlis quotient problem for a Hamiltonian-equivariant
  principal-part transfer retaining \(\mc P_{\le N}\) and killing its
  tail.  Inside the fixed-projection slice
  \(\mathsf{Curr}^{BM}_{\mathrm{fix}}\) of compatible
  finite-window principal-part current systems with full
  \(\mathfrak h\)-coadjoint action, BMK homotopy data, and uniform
  \(\mathfrak m\)-adic moment bounds
  (Theorem~\ref{thm:promatlis-universal-property}), the compatible
  replacement target used here is the one-pair analytic pro-Matlis
  retract
  \[
    \widehat{\mc P}_{\Pi}
      =\prod_{\substack{a,b\geq0\\a+b>0}}\C\rho_{a,b},
    \qquad
    \mc P_{\mathrm{an}}(B_1)
      =\tau\bigl(\mathcal O(B_1)^\vee_{\mathrm{red}}\bigr)
      \subset\widehat{\mc P}_{\Pi},
  \]
  with the normalized analytic contraction
  \(H_\chi^0=(1-I_{\mathrm{an}}P_{\mathrm{an}})H_\chi
  (1-I_{\mathrm{an}}P_{\mathrm{an}})\) of
  Theorem~\ref{thm:app-finite-window-bmk-current-transfer} satisfying
  \[
    \bar\partial H_\chi^0+H_\chi^0\bar\partial
      =\operatorname{id}-I_{\mathrm{an}}P_{\mathrm{an}},
  \]
  and the pro-analytic moment map
  \(P_{\Pi}^{\mathrm{an}}=\tau P_{\mathrm{an}}\) with compatible finite
  projections
  \[
    \pi_NP_{\Pi}^{\mathrm{an}}=p_N,
    \qquad
    \pi_{M,N}p_M=p_N
    \quad(M\geq N).
  \]
  This gives the projection-compatible inverse-limit receiver for the
  fixed BMK moment system.  Theorem~\ref{thm:bmk-finite-window-obstruction}
  establishes the necessity of a pro-Matlis replacement for this
  fixed-projection finite-window transfer problem.
\end{cor}

\begin{thm}[Pro-Matlis fixed-projection receiver criterion]\label{thm:promatlis-universal-property}
  Let \(\mathsf{Curr}^{BM}_{\mathrm{fix}}\) be the projection slice whose
  objects are
  \(\C\)-linear pairs
  \((Q,Q^{\mathrm{an}})\) with \(Q^{\mathrm{an}}\subset Q\), equipped
  with:
  \begin{enumerate}[label=(\alph*)]
    \item compatible projections \(\pi_N\colon Q\to\mathcal P_{\le N}\)
      satisfying \(\pi_N=\pi_{M,N}\circ\pi_M\) for \(M\ge N\);
    \item a Hamiltonian \(\mathfrak h\)-action through full residue-coadjoint
      operators on the untruncated map \(\mathfrak h\to\End(Q)\);
    \item a Dolbeault chain homotopy
      \(H\colon Q\to Q[-1]\) with \(\bar\partial H+H\bar\partial
       =\operatorname{id}-I\circ P\) for an analytic-submodule
      retraction
      \(P\colon Q\to Q^{\mathrm{an}},\ I\colon Q^{\mathrm{an}}
       \hookrightarrow Q\);
    \item a continuous analytic comparison
      \(\pi^{\mathrm{an}}\colon Q^{\mathrm{an}}\to
      \mathcal P_{\mathrm{an}}(B_1)\), compatible with the restrictions of
      the finite-window projections to the analytic submodule;
    \item uniform \(\mathfrak m\)-adic moment bounds: for every
      \(f\in\mathfrak h\) and every \(\xi\in Q\) of \(\mathfrak m\)-adic
      order \(\ge n\), \(f\cdot\xi\) has \(\mathfrak m\)-adic order
      \(\ge n-r(f)\) for an \(f\)-dependent constant \(r(f)\).
  \end{enumerate}
  Morphisms are \(\C\)-linear maps
  \((Q,Q^{\mathrm{an}})\to(Q',Q'^{\mathrm{an}})\) commuting with all
  finite projections \(\pi_N,\pi'_N\) and with the analytic-submodule
  retractions and analytic comparisons.  A morphism is
  \(\mathfrak h\)-equivariant only when the untruncated
  residue-coadjoint actions commute with it.  For every object
  \((Q,Q^{\mathrm{an}})\), the finite projections determine a unique
  \(\C\)-linear map
  \[
    \mu_Q\colon Q\longrightarrow
    \widehat{\mathcal P}_\Pi=\varprojlim_N\mathcal P_{\le N},
    \qquad
    \pi_N\mu_Q=\pi_N .
  \]
  Its restriction to \(Q^{\mathrm{an}}\) lands in
  \(\mathcal P_{\mathrm{an}}(B_1)\) exactly when the analytic
  compatibility equations
  \[
    \iota_{\mathrm{an}}\pi^{\mathrm{an}}=\mu_Q|_{Q^{\mathrm{an}}},
    \qquad
    \pi_N\iota_{\mathrm{an}}\pi^{\mathrm{an}}=\pi_N|_{Q^{\mathrm{an}}}
  \]
  hold.  It is a morphism in
  \(\mathsf{Curr}^{BM}_{\mathrm{fix}}\) only after a target analytic
  retraction
  \(P^{\mathrm{ret}}_\Pi\colon\widehat{\mathcal P}_\Pi\to
  \mathcal P_{\mathrm{an}}(B_1)\) with inclusion
  \(I_\Pi\colon\mathcal P_{\mathrm{an}}(B_1)\hookrightarrow
  \widehat{\mathcal P}_\Pi\) has been chosen and the retraction and
  Hamiltonian-action equations
  \[
    \mu_QIP=I_\Pi P^{\mathrm{ret}}_\Pi\mu_Q,\qquad
    \mu_Q(f\cdot\xi)=f\cdot\mu_Q(\xi)
  \]
  are satisfied for the chosen analytic retraction and for every
  \(f\in\mathfrak h\).  Here
  \(\iota_{\mathrm{an}}\colon\mathcal P_{\mathrm{an}}(B_1)
  \hookrightarrow\widehat{\mathcal P}_\Pi\) is the inclusion from the
  definition of \(\mathcal P_{\mathrm{an}}(B_1)\).  The assertion is
  therefore a fixed-projection
  receiver criterion for the BMK moment system, not a terminality
  statement after the category is enlarged by spectator summands, after
  the analytic submodule is changed, or before the displayed
  compatibility equations have been checked.
\end{thm}

\begin{proof}
  Existence: condition~(a) gives a compatible system
  \(\pi_N\colon Q\to\mathcal P_{\le N}\), defining a unique
  projection-compatible map
  \(Q\to\varprojlim_N\mathcal P_{\le N}=\widehat{\mathcal P}_\Pi\) by
  the universal property of the inverse limit.  Condition~(d), together
  with the displayed analytic compatibility equations, is exactly the
  statement that the restriction of this map to \(Q^{\mathrm{an}}\)
  factors through
  \(\mathcal P_{\mathrm{an}}(B_1)=
  \tau(\mathcal O(B_1)^\vee_{\mathrm{red}})\).  Condition~(e) gives the
  moment bound needed for convergence in the
  \(\mathfrak m\)-adic topology of \(\widehat{\mathcal P}_\Pi\).
  Uniqueness: any two such morphisms agree on every
  \(\mathcal P_{\le N}\)-component by compatibility with \(\pi_N\),
  hence on \(\varprojlim_N\mathcal P_{\le N}\).  The equations
  \(\mu_QIP=I_\Pi P^{\mathrm{ret}}_\Pi\mu_Q\) and
  \(\mu_Q(f\cdot\xi)=f\cdot\mu_Q(\xi)\) are additional structure, not
  consequences of the inverse limit.  Thus the theorem proves the
  projection-compatible receiver criterion and isolates the remaining
  homotopy and Hamiltonian-equivariance checks.
  Theorem~\ref{thm:bmk-finite-window-obstruction} excludes the fixed
  finite-rank quotient carrying the full Hamiltonian action, and
  Corollary~\ref{cor:promatlis-retract-compatible-replacement} verifies the
  pro-Matlis pair structure, the normalized Dolbeault contraction
  \(H^0_\chi\), and the analytic moment map \(P^{\mathrm{an}}_\Pi\)
  on \(\bigl(\widehat{\mathcal P}_\Pi,\mathcal P_{\mathrm{an}}(B_1)\bigr)\).
\end{proof}

\begin{proof}[Proof of Corollary~\ref{cor:promatlis-retract-compatible-replacement}]
  The first sentence is Theorem~\ref{thm:bmk-finite-window-obstruction}: the
  fixed finite-Matlis quotient has the Hamiltonian-action obstruction.  The
  positive constructions of the pro-Matlis pair \((\widehat{\mc
  P}_{\Pi},\mc P_{\mathrm{an}}(B_1))\), the normalized contraction
  \(H_\chi^0\), and the moment map \(P_{\Pi}^{\mathrm{an}}\) are
  Theorem~\ref{thm:app-finite-window-bmk-current-transfer}.

  Fixed-projection characterization.  An object in the stated
  \(\mathsf{Curr}^{BM}\) slice with a \(\C\)-linear
  \(\mathfrak h\)-module structure and a Dolbeault chain homotopy
  compatible with \(p_N\) and \(i_N\) at every \(N\) carries the
  following data:
  \begin{enumerate}[label=(\alph*)]
    \item accept the full Hamiltonian coadjoint action of
      \(\mathfrak h\),
    \item admit the Dolbeault homotopy obtained from the BMK kernel
      after the diagonal orientation and collar quotient,
    \item project compatibly to the finite Matlis windows used by the
      arity-two coefficient comparison
      \(\ell^{\mathrm{BM},N}_2(x,y)=p_N[ix,iy]\).
  \end{enumerate}
  Condition (a) excludes the fixed finite-rank quotient by
  Theorem~\ref{thm:bmk-finite-window-obstruction}; (b) selects the
  cohomology of the BMK contraction as the moment-restriction kernel,
  fixing the analytic submodule
  \(\mc P_{\mathrm{an}}(B_1)=\tau\bigl(\mathcal O(B_1)^\vee_{\mathrm{red}}\bigr)\)
  on which the kernel acts; (c) forces the fixed inverse-limit receiver
  for the finite projections to contain every \(\rho_{a,b}\) and to map
  to \(\mc P_{\leq N}\), hence to be
  \(\widehat{\mc P}_{\Pi}=\varprojlim_N\mc P_{\leq N}\) in this slice.
	  The pair \((\widehat{\mc P}_{\Pi},\mc P_{\mathrm{an}}(B_1))\) is
	  therefore the target used here with these fixed projections and BMK
  moment data.  The strict all-window factorization-current transfer
  requires the additional vanishing of
  \(\operatorname{Ob}^{\Pi}_{\mathrm{BM}}\) recorded in
  Theorem~\ref{thm:app-finite-window-bmk-current-transfer}, whose
  first three nonzero entries (direct-sum moment cokernel, missing
  compact-current section of the pro-product tower, divergent formal
  Hamiltonian action on the pro-product) are independent
  current-topology obstructions that remain alongside the analytic
  pro-Matlis retract.
\end{proof}

\begin{rmk}[Physical interpretation: high transverse holomorphic modes
  leak back into low brane momentum modes]\label{rmk:bmk-leak}
  The boundary-mode identity~\eqref{eq:bmk-bockstein-identity} of
  Theorem~\ref{thm:bmk-finite-window-obstruction} carries a direct physical
  reading.  In the Hamiltonian/Matlis dictionary the index
  \(\rho_{a,b}\) labels a holomorphic transverse mode of bidegree
  \((a,b)\) along \(\C^2_{\mathrm{hol}}\); the brane line
  \(L\subset\R^2_{\mathrm{top}}\times\C^2_{\mathrm{hol}}\) carries the
  low brane-momentum modes \(\rho_{0,b}\) and their finite Matlis
  truncations.  The class
  \[
    z_1^{N+2}\cdot\rho_{N+1,0}=-(N+2)\,\rho_{0,1}
  \]
  asserts that under Hamiltonian flow a high transverse holomorphic
  mode at degree \((N+1,0)\) -- formally above the cutoff window --
  transports back into a low brane-momentum mode at degree \((0,1)\)
  -- inside the cutoff window.  Physically: high transverse
  holomorphic modes leak back into low brane momentum modes.

  This leak is the obstruction for every strict finite-Matlis cutoff:
  truncating high transverse modes severs the Hamiltonian return
  channel that would have lifted them to brane momenta, but the
  return channel is already a finite-window operation by
  \eqref{eq:bmk-bockstein-identity}, so the image remains.  The
  pro-Matlis retract of
	  Corollary~\ref{cor:promatlis-retract-compatible-replacement} is the
	  fixed-projection module used here and closed under this leak: the full Taylor tower retains every
  high transverse mode and lets the Hamiltonian flow act
  consistently, while the analytic submodule
  \(\mc P_{\mathrm{an}}(B_1)\) gives the moment-restriction kernel
  required by the BMK contraction.  The arity-two projection
  \(\ell^{\mathrm{BM},N}_2(x,y)=p_N[ix,iy]\) of
  Theorem~\ref{thm:app-finite-window-bmk-current-transfer} is then a
  finite-window readout of this consistent flow; the finite-rank
  operation appears after this readout.
\end{rmk}
