\begin{abstract}

\noindent\textbf{Problem.}  Mixed holomorphic--topological closed
strings near a Dirac brane should coincide, at each formal
holomorphic-symplectic brane point, with the derived
\(E_1\)-centre / Hochschild cochain complex of the open boundary
algebra.

\medskip\noindent\textbf{Local model.}  On
\(X=\R^2_{\mathrm{top}}\times\C^2_{\mathrm{hol}}\), the closed sector
is the Hamiltonian BF theory with shifted-cotangent dg Lie input
\(\mathfrak g=\mathfrak h\ltimes\mathfrak h^\vee_{\mathrm{cont}}[1]\),
\(\mathfrak h=\C[[z_1,z_2]]/\C\cdot 1\), and Grothendieck--Matlis
residue dual.  The open sector is a stack of \(N\) Dirac branes with
matrix coordinates \((\phi_1,\phi_2)\in\mathfrak{gl}_N^{\oplus 2}\),
moment-map constraint \([\phi_1,\phi_2]=0\), antifield
\(\psi\in\mathfrak{gl}_N[1]\), and Koszul differential
\(Q\psi=[\phi_1,\phi_2]\); the classical brane algebra is
\(A^{\mathrm{cl}}_{\partial,N}=
C^\bullet_{\mathrm{CE}}\bigl(\mathfrak{gl}_N,\,
\mathrm{Sym}(\mathfrak{gl}_N\oplus\mathfrak{gl}_N\oplus\mathfrak{gl}_N[1])\bigr)\).

\medskip\noindent\textbf{Theorem.}  Inside the admissible filtered
nuclear Tate / pro-Matlis / Capelli-projective derived-centre category
\(\mathcal B^{\mathrm{adm}}\) (the canonical completion in which the
diagonal Casimir converges and the principal-part residue pairing is
perfect), the formal-Darboux stalk identification at \(N\) Dirac
branes is unconditional:
\[
  \mathcal A^{\mathrm{cl}}_{\mathrm{bulk}}\bigl|_{b}
   \;\simeq\;
  Z^{\mathrm{der,adm}}_{E_1}\!\bigl(A^{\mathrm{cl}}_{\partial,N}\bigr)
   \;\simeq\;
  HH^\bullet_{\mathrm{adm}}\bigl(A^{\mathrm{cl}}_{\partial,N},
                                 A^{\mathrm{cl}}_{\partial,N}\bigr)
\]
as \(E_2\)-algebras, with formal Darboux coordinate
\(J(f)=\overline{\operatorname{Tr}\,f(\phi_1,\phi_2)}\) and CE/PV
dictionary \(c_f\mapsto\theta_f\), \(u_f\mapsto J(f)\).  The
finite-\(N\) scalar anomaly is the Capelli central charge
\(\hbar N[\bar c]=\Omega_{\mathrm{central}}|_{J(f)}\), the projective
curvature of the modular line bundle at the trace generator.

\medskip\noindent\textbf{Boundary.}  Strict-product, unreduced
cotangent, and global lifts are governed by named obstruction classes
(Tate / pro-Matlis admissibility, weighted-Casimir convergence,
filtered cobar acceptance, all-loop QME, descent on
\(\partial X\)).  Compact targets and global modular claims (chiral
\(E_d\) Deligne at \(d\geq 2\); Hall--Drinfeld--Borcherds on compact
non-toric CY\(_3\); operator-level lift of \(\Phi_{10}\) on
\(K3\times E\); \(W_\infty[\lambda]/E_\infty\) admissible endpoint) are
explicitly conditional, with elaboration in
\texttt{working\_notes/frontier.tex}.

\end{abstract}
